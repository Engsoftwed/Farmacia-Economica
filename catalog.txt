/

Helper de categorização inteligente no Front-end.

Corrige falhas onde medicamentos eram rotulados como 'HIGIENE'.
*/
export function classifyProduct(product) {
const name = (product.produto || '').toLowerCase();

// Lista de palavras-chave de Higiene e Perfumaria
const perfumariaKeywords = [
'desodorante', 'absorvente', 'sabonete', 'shampoo', 'condicionador',
'creme', 'fralda', 'protetor', 'higiene', 'colonia', 'perfume',
'esponja', 'lamina', 'lâmina', 'talco', 'batom', 'esmalte', 'lenco', 'lenço'
];

const isPerfumaria = perfumariaKeywords.some(keyword => name.includes(keyword));

if (isPerfumaria) {
return {
categoria: 'Perfumaria',
percentualDesconto: 12,
tipoMedicamento: null
};
}

// Se não for perfumaria, é Medicamento
const isGeneric = name.includes('gen') || name.includes('med.gen') || name.includes('generico');

return {
categoria: 'Medicamento',
percentualDesconto: 10,
tipoMedicamento: isGeneric ? 'Genérico/Similar' : 'Original/Referência'
};
}
